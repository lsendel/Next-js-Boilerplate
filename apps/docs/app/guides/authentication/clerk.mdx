# Clerk Authentication

Complete guide to using Clerk as your authentication provider.

## Overview

Clerk is the default authentication provider in the boilerplate with:

- **Pre-configured** - Ready to use out of the box
- **Keyless Development** - No API keys needed locally
- **Multi-Language** - Full i18n support
- **Social Login** - Google, GitHub, Microsoft, etc.
- **MFA** - Two-factor authentication
- **Session Management** - Secure session handling
- **User Management** - Built-in user admin

## Quick Start

### 1. Development Mode (Keyless)

```bash
# No configuration needed!
pnpm --filter web dev

# Visit: http://localhost:3000/sign-in
```

Clerk works out of the box in development using test keys from `.env`.

### 2. Production Setup

1. **Create Clerk account** at [clerk.com](https://clerk.com)
2. **Create application**
3. **Copy API keys**:
   - Publishable key
   - Secret key

4. **Add to environment:**

```bash
# apps/web/.env.local
NEXT_PUBLIC_CLERK_PUBLISHABLE_KEY=pk_test_...
CLERK_SECRET_KEY=sk_test_...
```

## Configuration

### Clerk Provider Setup

```typescript
// src/libs/auth/adapters/ClerkAdapter.tsx
import { ClerkProvider as Provider } from '@clerk/nextjs'
import { AppConfig } from '@/utils/AppConfig'

export function ClerkProvider({ children }: { children: React.ReactNode }) {
  return (
    <Provider
      localization={getClerkLocale()}
      appearance={{
        elements: {
          formButtonPrimary: 'bg-blue-600 hover:bg-blue-700',
        },
      }}
    >
      {children}
    </Provider>
  )
}

function getClerkLocale() {
  const locale = useLocale()

  const localeMap = {
    en: 'en-US',
    fr: 'fr-FR',
    es: 'es-ES',
    de: 'de-DE',
  }

  return localeMap[locale] || 'en-US'
}
```

### API Route Protection

```typescript
// app/api/users/route.ts
import { auth } from '@clerk/nextjs/server'
import { NextResponse } from 'next/server'

export async function GET(request: Request) {
  const { userId } = await auth()

  if (!userId) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
  }

  // Fetch user data
  const users = await getUsers(userId)

  return NextResponse.json(users)
}
```

## Client-Side Auth

### useAuth Hook

```typescript
'use client'

import { useAuth } from '@clerk/nextjs'

export function ProtectedComponent() {
  const { isLoaded, userId, sessionId, getToken } = useAuth()

  if (!isLoaded) {
    return <div>Loading...</div>
  }

  if (!userId) {
    return <div>Please sign in</div>
  }

  return <div>Welcome! User ID: {userId}</div>
}
```

### useUser Hook

```typescript
'use client'

import { useUser } from '@clerk/nextjs'

export function UserProfile() {
  const { isLoaded, isSignedIn, user } = useUser()

  )
```

## Middleware Protection

```typescript
// src/middleware.ts
import { clerkMiddleware, createRouteMatcher } from '@clerk/nextjs/server'

const isProtectedRoute = createRouteMatcher([
  '/dashboard(.*)',
  '/settings(.*)',
  '/profile(.*)',
])

export default clerkMiddleware((auth, req) => {
  if (isProtectedRoute(req)) {
    auth().protect()
  }
})

export const config = {
  matcher: ['/((?!.*\\..*|_next).*)', '/', '/(api|trpc)(.*)'],
}
```

## Session Management

### Session Duration

Configure in Clerk Dashboard:
- **Session lifetime:** 7 days (default)
- **Inactivity timeout:** 30 minutes
- **Multi-session:** Allowed

### Get Session

```typescript
'use client'

import { useSession } from '@clerk/nextjs'

export function SessionInfo() {
  const { session, isLoaded } = useSession()

  if (!isLoaded) return <div>Loading...</div>

  return (
    <div>
      <p>Session ID: {session?.id}</p>
      <p>Expires: {session?.expireAt.toLocaleString()}</p>
    </div>
  )
}
```

### Session Tokens

```typescript
'use client'

import { useAuth } from '@clerk/nextjs'

export function useApiCall() {
  const { getToken } = useAuth()

  const callAPI = async () => {
    const token = await getToken()

    const response = await fetch('/api/data', {
      headers: {
        Authorization: `Bearer ${token}`,
      },
    })

    return response.json()
  }

  return { callAPI }
}
```

## Webhooks

### Setup Webhooks

1. **Clerk Dashboard → Webhooks**
2. **Add endpoint:** `https://yourapp.com/api/webhooks/clerk`
3. **Select events:**
   - user.created
   - user.updated
   - user.deleted
   - session.created

### Handle Webhooks

```typescript
// app/api/webhooks/clerk/route.ts
import { Webhook } from 'svix'
import { headers } from 'next/headers'

export async function POST(request: Request) {
  const WEBHOOK_SECRET = process.env.CLERK_WEBHOOK_SECRET

  const headerPayload = headers()
  const svixId = headerPayload.get('svix-id')
  const svixTimestamp = headerPayload.get('svix-timestamp')
  const svixSignature = headerPayload.get('svix-signature')

  const body = await request.text()

  const wh = new Webhook(WEBHOOK_SECRET!)
  const evt = wh.verify(body, {
    'svix-id': svixId!,
    'svix-timestamp': svixTimestamp!,
    'svix-signature': svixSignature!,
  })

  const { type, data } = evt

  switch (type) {
    case 'user.created':
      await handleUserCreated(data)
      break
    case 'user.updated':
      await handleUserUpdated(data)
      break
    case 'user.deleted':
      await handleUserDeleted(data)
      break
  }

  return new Response('', { status: 200 })
}

async function handleUserCreated(data: any) {
  // Sync user to your database
  await db.insert(users).values({
    id: data.id,
    email: data.email_addresses[0].email_address,
    name: `${data.first_name} ${data.last_name}`,
  })
}
```
